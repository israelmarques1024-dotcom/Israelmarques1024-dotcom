function math(id,g,t){
 const a=mod(id,9000,11)+2,b=mod(id,8000,13)+2,c=mod(id,500,17)+2;
 if(t.includes('Fraç')){
  const n1=a%97+1,d1=b%89+2,n2=c%83+1,d2=mod(id,79,19)+2;
  const num=n1*d2+n2*d1, den=d1*d2, gcd=(x,y)=>{while(y){[x,y]=[y,x%y]}return x},z=gcd(num,den);
  return rec(id,g,'Matemática',t,`Calcule e simplifique ${n1}/${d1} + ${n2}/${d2}.`,`${num/z}/${den/z}`,'Use denominador comum e simplifique.','Multiplique cada numerador pelo denominador da outra fração.');
 }

  return rec(id,g,'Matemática',t,`Quanto é ${String(p).replace('.',',')}% de ${v}?`,fmt(ans),`${p}/100 × ${v} = ${fmt(ans)}.`,'Transforme a porcentagem em uma razão por 100.');

  }
  const rhs=k*x+m;
  return rec(id,g,'Matemática',t,`Resolva: ${k}x + (${m}) = ${rhs}.`,x,`Isolando x, obtemos x=${x}.`,'Desfaça soma/subtração e depois a multiplicação.');
 }

  const total=a%90+10,fav=b%total;
  return rec(id,g,'Matemática',t,`Há ${total} resultados equiprováveis e ${fav} favoráveis. Qual é a probabilidade?`,`${fav}/${total}`,`P=${fav}/${total}.`,'Casos favoráveis ÷ casos possíveis.');
 }
 if(t.includes('Geometr')||t.includes('Plano')||t.includes('Semelhan')||t.includes('Radicia')){
  const w=a%100+2,h=b%100+2;
  return rec(id,g,'Matemática',t,`Qual é a área de um retângulo de ${w} cm por ${h} cm?`,`${w*h} cm²`,`A=${w}×${h}=${w*h} cm².`,'Base × altura.');

 return rec(id,g,'Matemática',t,`Calcule ${x} ${op} (${y}).`,ans,`O resultado é ${ans}.`,'Observe os sinais e a ordem da operação.');
}
