function generate(id){
 id=Number(id);
 if(!Number.isInteger(id)||id<0||id>=TOTAL)throw new Error('ID deve estar entre 0 e 99.999.999');
 const g=gradeFor(id),s=subjectFor(id),t=topicFor(id,g,s);
 return s==='Matemática'?math(id,g,t):s==='Língua Portuguesa'?portuguese(id,g,t):s==='Ciências'?science(id,g,t):s==='História'?history(id,g,t):s==='Geografia'?geography(id,g,t):s==='Inglês'?english(id,g,t):programming(id,g,t);
}

function randomId(grade,subject){
 let start=(grade-6)*PER_GRADE;
 for(let tries=0;tries<100;tries++){const id=start+Math.floor(Math.random()*PER_GRADE);if(subjectFor(id)===subject)return id}
 return start+SUBJECTS.indexOf(subject);
}
