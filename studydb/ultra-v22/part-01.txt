&p!==document.body;i++,p=p.parentElement){
    const radios=p.querySelectorAll('input[type="radio"],input[type="checkbox"],[role="radio"]').length;
    const txt=cleanText(p).length;
    if(radios===optionElements.length&&txt<=6000)best=p;
    else break;
  }
  return best;
}

function optionFromElement(el){
  const visual=el.matches?.('input[type="radio"],input[type="checkbox"]')?labelForInput(el):el;
  let text=cleanText(visual);
  if(!text&&el.getAttribute)text=el.getAttribute('aria-label')||el.getAttribute('value')||'';
  return {element:el,visualElement:visual,text:text.trim().slice(0,800)};
}

function extractQuestionText(root,options){
  if(!root)return '';
  const clone=root.cloneNode(true);

  // Remove as próprias alternativas e elementos de ação para sobrar o enunciado.
  const texts=new Set(options.map(o=>normalize(o.text)));
  [...clone.querySelectorAll('input,button,select,textarea,script,style')].forEach(x=>x.remove());

  // Remove labels/opções cujo texto bate com alternativas.
  [...clone.querySelectorAll('label,[role="radio"],[role="option"],[class*="option" i],[class*="choice" i]')].forEach(x=>{
    const t=normalize(cleanText(x));
    if(t&&texts.has(t))x.remove();
  });

  let q=cleanText(clone);

  // Fallback: usa texto total e remove cada alternativa uma vez.
  if(q.length<8){
    q=cleanText(root);
    for(const o of options){
      const pos=q.indexOf(o.text);
      if(pos>=0)q=q.slice(0,pos)+q.slice(pos+o.text.length);
    }
    q=q.replace(/\s+/g,' ').trim();
  }

  // Remove ruído visual típico sem destruir o enunciado.
  q=q.replace(/Quest[aã]o\s*\d+\s*de\s*\d+/ig,' ')
     .replace(/\b\d+\s*PONTOS?\b/ig,' ')
     .replace(/\s+/g,' ').trim();

  return q.slice(0,2200);
}

function getAllQuestions(){
  const all=[];
  const seen=new Set();
  const groups=[...radioGroups(),...roleRadioGroups()];

  for(const group of groups){
    const options=group.map(optionFromElement).filter(o=>o.text&&o.text.length<800);
    if(options.length<2)continue;

    const root=chooseQuestionRoot(group);
    if(!root||!visible(root))continue;
    const question=extractQuestionText(root,options);
    if(question.length<8)continue;

    const signature=normalize(question)+'|'+options.map(o=>normalize(o.text)).join('|');
    if(seen.has(signature))continue;
    seen.add(signature);

    all.push({
      container:root,question,options,platform:platformName(),
      status:'pending',hash:null,suggestion:null,confidence:0,reason:''
    });
    if(all.length>=settings().maxQuestions)break;
  }
  return all;
}

function optionIndexByAnswer(options,answer){
  const target=normalize(answer);
  if(!target)return -1;
  let i=options.findIndex(o=>normalize(o.text)===target);
  if(i>=0)return i;
  return options.findIndex(o=>{
    const t=normalize(o.text);
    return target.length>=2&&(t.includes(target)||target.includes(t));
  });
}

function numericValue(s){
  const x=String(s).trim().replace(/\s/g,'').replace(',','.').replace(/[^\d.+\-]/g,'');
  const n=Number(x);return Number.isFinite(n)?n:null;
}
function formattedNumbers(n){
  if(!Number.isFinite(n))return [];
  const xs=new Set([String(n),String(n).replace('.',','),String(Math.round(n*1000)/1000),String(Math.round(n*1000)/1000).replace('.',',')]);
  return [...xs];
}
function matchNumericOption(options,value,unitPattern=''){
  for(const v of formattedNumbers(value)){
    let idx=options.findIndex(o=>{
      const t=normalize(o.text);
      if(!t.includes(normalize(v)))return false;
      return !unitPattern||new RegExp(unitPattern,'i').test(t);
    });
    if(idx>=0)return idx;
  }
  // Último recurso: compara o primeiro número da alternativa.
  for(let i=0;i<options.length;i++){
    const m=options[i].text.match(/-?\d+(?:[.,]\d+)?/);
    if(m&&Math.abs((numericValue(m[0])??Infinity)-value)<1e-9)return i;
  }
  return -1;
}
