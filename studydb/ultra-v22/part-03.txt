   question:entry.question,
    answer:entry.options[index].text,
    optionIndex:index,
    options:entry.options.map(o=>o.text),
    platform:entry.platform,
    createdAt:prev?.createdAt||now,updatedAt:now,
    revision:(prev?.revision||0)+1,source:'manual-confirmation'
  };
  saveDB(db);
  entry.suggestion={index,answer:entry.options[index].text,confidence:1,reason:'Resposta confirmada no banco local',source:'manual-confirmation'};
  entry.status='found';entry.reason=entry.suggestion.reason;
  markAll();render();
}

function exportDB(){
  const a=document.createElement('a');
  const blob=new Blob([JSON.stringify({version:VERSION,exportedAt:new Date().toISOString(),records:loadDB()},null,2)],{type:'application/json'});
  a.href=URL.createObjectURL(blob);a.download=`doritus-study-bank-${new Date().toISOString().slice(0,10)}.json`;a.click();
  setTimeout(()=>URL.revokeObjectURL(a.href),1000);
}
function importDB(file){
  const r=new FileReader();
  r.onload=()=>{
    try{
      const d=JSON.parse(r.result),incoming=d.records||d,db=loadDB();let n=0;
      for(const [h,x] of Object.entries(incoming)){
        if(!/^[a-f0-9]{64}$/i.test(h)||!x?.question||x.answer==null)continue;
        db[h]=x;n++;
      }
      saveDB(db);alert(`${n} registros importados.`);analyzeAll();
    }catch(e){alert('Falha ao importar: '+e.message)}
  };
  r.readAsText(file);
}
