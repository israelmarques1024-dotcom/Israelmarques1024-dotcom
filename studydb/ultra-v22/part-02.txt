(\d+(?:[.,]\d+)?)\s*cm.*(?:termina|at[eé])(?:\s+na)?\s*(?:marca\s*)?de?\s*(\d+(?:[.,]\d+)?)\s*cm/i);
  if(m){
    const a=numericValue(m[1]),b=numericValue(m[2]),v=Math.abs(b-a),idx=matchNumericOption(options,v,'cm');
    if(idx>=0)return {index:idx,answer:options[idx].text,confidence:.99,reason:`Comprimento = ${b} − ${a} = ${v} cm`,source:'local-solver'};
  }

  // Aritmética.
  m=q.match(/(?:calcule|resultado|valor de)?\s*(-?\d+(?:[.,]\d+)?)\s*([+\-×x*/÷])\s*\(?\s*(-?\d+(?:[.,]\d+)?)\s*\)?/i);
  if(m){
    const a=numericValue(m[1]),b=numericValue(m[3]),op=m[2];
    let v=null;
    if(op==='+')v=a+b; else if(op==='-')v=a-b;
    else if(op==='×'||op.toLowerCase()==='x'||op==='*')v=a*b;
    else if((op==='/'||op==='÷')&&b!==0)v=a/b;
    if(Number.isFinite(v)){
      const idx=matchNumericOption(options,v);
      if(idx>=0)return {index:idx,answer:options[idx].text,confidence:.97,reason:`Operação calculada = ${v}`,source:'local-solver'};
    }
  }

  // Porcentagem.
  m=q.match(/(\d+(?:[.,]\d+)?)\s*%\s*(?:de|do|da)\s*(\d+(?:[.,]\d+)?)/i);
  if(m){
    const p=numericValue(m[1]),b=numericValue(m[2]),v=p*b/100,idx=matchNumericOption(options,v);
    if(idx>=0)return {index:idx,answer:options[idx].text,confidence:.97,reason:`${p}% de ${b} = ${v}`,source:'local-solver'};
  }

  // Equação ax+b=c.
  m=q.match(/(-?\d*)\s*x\s*([+\-])\s*(\d+(?:[.,]\d+)?)\s*=\s*(-?\d+(?:[.,]\d+)?)/i);
  if(m){
    const a=m[1]===''||m[1]==='+'?1:m[1]==='-'?-1:Number(m[1]);
    const b=numericValue(m[3]),c=numericValue(m[4]);
    const x=m[2]==='-'?(c+b)/a:(c-b)/a;
    const idx=matchNumericOption(options,x);
    if(idx>=0)return {index:idx,answer:options[idx].text,confidence:.95,reason:`Equação resolvida: x=${x}`,source:'local-solver'};
  }

  // Regras conceituais locais.
  const rules=[
    [/bact[eé]rias?.*procarion|procariontes?.*bact[eé]r/i,/procarion/i,'Bactérias são procariontes.'],
    [/estrutura.*entrada.*sa[ií]da.*subst[aâ]ncias.*c[eé]lula/i,/membrana plasm/i,'Membrana plasmática controla trocas.'],
    [/unidade.*for[cç]a.*si/i,/newton|\bn\b/i,'Força é medida em newtons.'],
    [/simple past.*\bgo\b/i,/went/i,'go → went'],
    [/simple past.*\bsee\b/i,/saw/i,'see → saw']
  ];
  for(const [qr,ar,why] of rules){
    if(qr.test(q)){
      const idx=options.findIndex(o=>ar.test(normalize(o.text)));
      if(idx>=0)return {index:idx,answer:options[idx].text,confidence:.86,reason:why,source:'local-solver'};
    }
  }
  return null;
}

async function lookup(entry){
  const db=loadDB(),hash=await sha256(entry.question); entry.hash=hash;
  const exact=db[hash];
  if(exact){
    const idx=Number.isInteger(exact.optionIndex)?exact.optionIndex:optionIndexByAnswer(entry.options,exact.answer);
    if(idx>=0&&idx<entry.options.length)
      return {index:idx,answer:entry.options[idx].text,confidence:1,reason:'Resposta confirmada no banco local',source:'local-exact'};
  }

  let best=null;
  for(const [key,r] of Object.entries(db)){
    if(!r?.question)continue;
    const score=jaccard(entry.question,r.question);
    if(score>=settings().fuzzyThreshold&&(!best||score>best.score))best={key,r,score};
  }
  if(best){
    const idx=Number.isInteger(best.r.optionIndex)?best.r.optionIndex:optionIndexByAnswer(entry.options,best.r.answer);
    if(idx>=0&&idx<entry.options.length)
      return {index:idx,answer:entry.options[idx].text,confidence:best.score,reason:`Banco local aproximado (${Math.round(best.score*100)}%)`,source:'local-fuzzy'};
  }

  return solveSimple(entry.question,entry.options);
}

function clearMarks(){
  document.querySelectorAll('.'+HIGHLIGHT_CLASS).forEach(x=>x.classList.remove(HIGHLIGHT_CLASS));
  document.querySelectorAll('.'+BADGE_CLASS).forEach(x=>x.remove());
}

function markEntry(entry){
  if(!entry?.suggestion||!Number.isInteger(entry.suggestion.index))return;
  const o=entry.options[entry.suggestion.index],v=o?.visualElement;
  if(!v)return;
  v.classList.add(HIGHLIGHT_CLASS);

  // Uma etiqueta pequena junto da alternativa, sem clicar nela.
  if(!v.querySelector?.('.'+BADGE_CLASS)){
    const b=document.createElement('span');
    b.className=BADGE_CLASS;
    b.textContent='Doritus: sugestão';
    try{v.appendChild(b)}catch{}
  }
}
function markAll(){
  clearMarks();
  state.scan.filter(x=>x.status==='found').forEach(markEntry);
}

async function analyzeAll(){
  if(state.scanning)return;
  state.scanning=true;
  try{
    clearMarks();
    state.scan=getAllQuestions();
    for(const e of state.scan){
      const r=await lookup(e);
      if(r){e.status='found';e.suggestion=r;e.confidence=r.confidence;e.reason=r.reason}
      else{e.status='missing';e.reason='Sem resposta no banco local e sem solução local confiável.'}
    }
    markAll();
    render();
  }finally{state.scanning=false}
}
