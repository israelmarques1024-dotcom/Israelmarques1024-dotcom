function normalize(text){
  return String(text||'').normalize('NFKC').toLocaleLowerCase('pt-BR')
    .replace(/\u00a0/g,' ').replace(/\s+/g,' ')
    .replace(/[“”"'`´]/g,'').replace(/\s*([,.;:!?()[\]{}])\s*/g,'$1')
    .trim().slice(0,3000);
}
async function sha256(text){
  const d=await crypto.subtle.digest('SHA-256',new TextEncoder().encode(normalize(text)));
  return [...new Uint8Array(d)].map(x=>x.toString(16).padStart(2,'0')).join('');
}
function tokenize(t){return new Set(normalize(t).split(/[^\p{L}\p{N}]+/u).filter(x=>x.length>1))}
function jaccard(a,b){
  const A=tokenize(a),B=tokenize(b); if(!A.size||!B.size)return 0;
  let c=0; for(const x of A)if(B.has(x))c++;
  return c/(A.size+B.size-c);
}

function esc(s){

function commonAncestor(elements){
  if(!elements.length)return null;
  let a=elements[0];
  while(a&&a!==document.documentElement){
    if(elements.every(e=>a.contains(e)))return a;
    a=a.parentElement;
  }
  return document.body;
}

function labelForInput(input){
  if(input.id){
    try{
      const l=document.querySelector(`label[for="${CSS.escape(input.id)}"]`);
      if(l)return l;
    }catch{}
  }
  const own=input.closest('label');
  if(own)return own;

  // Bootstrap/custom radios frequentemente deixam o texto no pai/irmão.
  let p=input.parentElement;
  for(let i=0;i<4&&p;i++,p=p.parentElement){
    const t=cleanText(p);
    if(t&&t.length<=600)return p;
  }
  return input;
}

function radioGroups(){
  const inputs=[...document.querySelectorAll('input[type="radio"],input[type="checkbox"]')].filter(visible);
  const groups=new Map();
  for(const input of inputs){
    const name=input.getAttribute('name')||input.dataset.dusAnonName||(input.dataset.dusAnonName='dus-'+Math.random().toString(36).slice(2));
    const key=(input.type||'radio')+'|'+name;
    if(!groups.has(key))groups.set(key,[]);
    groups.get(key).push(input);
  }
  return [...groups.values()].filter(g=>g.length>=2&&g.length<=20);
}

function roleRadioGroups(){

  for(const r of radios){

    if(!root)continue;

    byRoot.get(root).push(r);
  }
  return [...byRoot.values()].filter(g=>g.length>=2&&g.length<=20);
}
